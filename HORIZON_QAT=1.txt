@torch.no_grad()
def horizon_calibration(
    calib_model,
    data_loader,
    example_input,
    num_iters=6,
):
    calib_model.eval()

    set_fake_quantize(
        calib_model,
        FakeQuantState.CALIBRATION,
    )

    calib_iter = 0

    for raw_data in data_loader:

        if calib_iter >= num_iters:
            break

        data = _extract_from_dc(raw_data)

        required_keys = [
            "img",
            "cam_extrinsic",
            "cam_intrinsic",
            "cam_distortion",
            "image_wh",
        ]

        missing_keys = [
            key
            for key in required_keys
            if key not in data
        ]

        if missing_keys:
            raise KeyError(
                "Sparse4D calibration data missing keys: "
                f"{missing_keys}. "
                f"Available keys: {list(data.keys())}"
            )

        img = data["img"]
        if isinstance(img, list):
            assert len(img) == 1, \
                f"Unexpected img list length: {len(img)}"
            img = img[0]
        assert img.ndim == 5, (
            f"Expected img shape "
            f"(B,Ncam,C,H,W), got {img.shape}"
        )
        assert torch.isfinite(
            img
        ).all()

        cam_extrinsic = data["cam_extrinsic"]
        cam_intrinsic = data["cam_intrinsic"]
        cam_distortion = data["cam_distortion"]
        image_wh = data["image_wh"]

        assert torch.isfinite(
            cam_extrinsic
        ).all()

        assert torch.isfinite(
            cam_intrinsic
        ).all()

        assert torch.isfinite(
            cam_distortion
        ).all()

        assert torch.isfinite(
            image_wh
        ).all()

        time_interval = _tile_to_batch(
            example_input[5],
            img.shape[0],
        )
        
        if calib_cached_feature is not None:
            calib_cached_feature = _tile_to_batch(
                calib_cached_feature,
                img.shape[0],
            )
        else:
            calib_cached_feature = None


        if calib_cached_anchor is not None:
            calib_cached_anchor = _tile_to_batch(
                calib_cached_anchor,
                img.shape[0],
            )
        else:
            calib_cached_anchor = None

        if calib_cache_confidence is not None:
            calib_cache_confidence = _tile_to_batch(
                calib_cache_confidence,
                img.shape[0],
            )
        else:
            calib_cache_confidence = None


        if calib_cache_track_id is not None:
            calib_cache_track_id = _tile_to_batch(
                calib_cache_track_id,
                img.shape[0],
            )
        else:
            calib_cache_track_id = None

        print(
            "[QAT-04 DEBUG] img type:",
            type(img),
            "shape:",
            img.shape,
        )

        det_output = _run_det_forward(
            calib_model,
            img,
            cam_extrinsic,
            cam_intrinsic,
            cam_distortion,
            image_wh,
            time_interval,
        )
        if not isinstance(det_output, dict):
            raise RuntimeError(
                "Sparse4D horizon_forward must return dict, "
                f"but got {type(det_output)}"
            )

        calib_iter += 1

        print(
            f"calibration iter : "
            f"{calib_iter}/{num_iters}"
        )
